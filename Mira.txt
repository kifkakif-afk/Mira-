                    orders = JSON.parse(e.newValue || '[]');
                    renderAdminOrders();

                }

        // Broadcast changes to other open tabs
        function broadcastChange(type, payload = {}) {
            const syncData = { type, payload, timestamp: Date.now() };
            try {
                realtimeChannel.postMessage(syncData);
            } catch (err) {
                console.log('BroadcastChannel sync error:', err);
            }
        }

        // Handle incoming realtime signals
        function handleRealtimeUpdate(data) {
            if (data.type === 'PACKAGES_UPDATED') {
                loadPackagesFromStorage();
                renderCustomerPackages();
                renderAdminPackages();
                showToast('🟢 ราคาแพ็กเกจถูกอัปเดตแบบ Realtime จากแอดมิน!', 'info');
            } else if (data.type === 'ORDER_CREATED') {
                loadOrdersFromStorage();
                renderAdminOrders();
                updatePendingOrdersBadge();
                showToast('📥 มีคำสั่งซื้อใหม่เข้ามาในระบบ!', 'success');
            } else if (data.type === 'ORDER_STATUS_CHANGED') {
                loadOrdersFromStorage();
                renderAdminOrders();
                showToast(`คำสั่งซื้อ #${data.payload.orderId.slice(-6)} เปลี่ยนสถานะเป็น ${data.payload.status}`, 'info');
            }
        }

        // --- STORAGE HELPERS ---
        function loadPackagesFromStorage() {
            const stored = localStorage.getItem('mira_rov_packages');
            if (stored) {
                packages = JSON.parse(stored);
            } else {
                packages = [...INITIAL_PACKAGES];
                savePackagesToStorage(false);
            }
        }

        function savePackagesToStorage(notify = true) {
            localStorage.setItem('mira_rov_packages', JSON.stringify(packages));
            if (notify) {
                broadcastChange('PACKAGES_UPDATED');
            }
        }

        function loadOrdersFromStorage() {
            const stored = localStorage.getItem('mira_rov_orders');
            orders = stored ? JSON.parse(stored) : [];
            updatePendingOrdersBadge();
        }

        function saveOrdersToStorage(notify = true) {
            localStorage.setItem('mira_rov_orders', JSON.stringify(orders));
            updatePendingOrdersBadge();
            if (notify) {
                broadcastChange('ORDER_CREATED');
            }
        }

        function submitOrder() {
            if (!verifiedPlayer) {
                showToast('กรุณากรอก Open ID และกดตรวจสอบชื่อตัวละครใน Step 1 ก่อนครับ', 'error');

                return;

            const newOrder = {
                id: 'ORD-' + Date.now(),
                timestamp: new Date().toLocaleString('th-TH'),
                openId: verifiedPlayer.openId,
                playerName: verifiedPlayer.name,
                playerRank: verifiedPlayer.rank,
                packageName: pkg.name,
                coupons: pkg.coupons,
                price: pkg.price,
                slipImage: uploadedSlipBase64,
                status: 'Pending' // Pending, Approved, Rejected

            document.getElementById('modalPkgId').value = pkg.id;
            document.getElementById('modalPkgName').value = pkg.name;
            document.getElementById('modalPkgCoupons').value = pkg.coupons;
            document.getElementById('modalPkgPrice').value = pkg.price;
            document.getElementById('modalPkgActive').value = pkg.active ? 'true' : 'false';
            document.getElementById('packageModal').classList.remove('hidden');
        }

        function closePackageModal() {
            document.getElementById('packageModal').classList.add('hidden');
        }

        function savePackageForm(e) {
            e.preventDefault();
            const id = document.getElementById('modalPkgId').value;
            const name = document.getElementById('modalPkgName').value.trim();
            const coupons = parseInt(document.getElementById('modalPkgCoupons').value);
            const price = parseFloat(document.getElementById('modalPkgPrice').value);
            const active = document.getElementById('modalPkgActive').value === 'true';

            if (id) {
                // Update Existing
                const pkg = packages.find(p => p.id === id);
                if (pkg) {
                    pkg.name = name;
                    pkg.coupons = coupons;
                    pkg.price = price;
                    pkg.active = active;
                }
            } else {
                // Add New
                const newPkg = {
                    id: 'pkg_' + Date.now(),
                    name,
                    coupons,
                    price,
                    active
                };
                packages.push(newPkg);
            }

            savePackagesToStorage(true);
            renderAdminPackages();
            renderCustomerPackages();
            closePackageModal();
            showToast('บันทึกข้อมูลแพ็กเกจและอัปเดตแบบ Realtime แล้ว!', 'success');
        }

        // Render Admin Orders
        function renderAdminOrders() {
            const tbody = document.getElementById('adminOrdersTable');
            tbody.innerHTML = '';

            if (orders.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="text-center p-6 text-gray-500">ยังไม่มีรายการสั่งซื้อในขณะนี้</td></tr>`;
                return;
            }

            orders.forEach(ord => {
                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-900/50 transition-colors";

                let statusBadge = '';
                if (ord.status === 'Pending') {
                    statusBadge = `<span class="bg-amber-500/20 text-amber-400 border border-amber-500/30 px-2 py-0.5 rounded text-[10px]">รอดำเนินการ</span>`;
                } else if (ord.status === 'Approved') {
                    statusBadge = `<span class="bg-emerald-500/20 text-emerald-400 border border-emerald-500/30 px-2 py-0.5 rounded text-[10px]">อนุมัติแล้ว</span>`;
                } else {
                    statusBadge = `<span class="bg-red-500/20 text-red-400 border border-red-500/30 px-2 py-0.5 rounded text-[10px]">ปฏิเสธ</span>`;
                }

                tr.innerHTML = `
                    <td class="p-3">
                        <p class="font-mono text-[11px] text-gray-400">${ord.id}</p>
                        <p class="text-[10px] text-gray-500">${ord.timestamp}</p>
                    </td>
                    <td class="p-3">
                        <p class="font-bold text-white">${ord.playerName}</p>
                        <p class="font-mono text-[10px] text-amber-400">${ord.openId}</p>
                    </td>
                    <td class="p-3 font-semibold text-cyan-400">${ord.packageName}</td>
                    <td class="p-3 font-orbitron text-amber-400 font-bold">฿${ord.price.toFixed(2)}</td>
                    <td class="p-3 text-center">
                        <button onclick="viewSlipModal('${ord.slipImage}')" class="px-2 py-1 bg-esports-bg hover:bg-slate-700 border border-esports-border rounded text-[11px] text-gray-300">
                            <i class="fa-solid fa-image text-amber-400"></i> ดูสลิป
                        </button>
                    </td>
                    <td class="p-3 text-center">${statusBadge}</td>
                    <td class="p-3 text-right space-x-1">
                        ${ord.status === 'Pending' ? `
                            <button onclick="updateOrderStatus('${ord.id}', 'Approved')" class="px-2 py-1 bg-emerald-600/30 hover:bg-emerald-600/60 text-emerald-300 rounded text-[11px]">
                                อนุมัติ
                            </button>
                            <button onclick="updateOrderStatus('${ord.id}', 'Rejected')" class="px-2 py-1 bg-red-600/30 hover:bg-red-600/60 text-red-300 rounded text-[11px]">
                                ปฏิเสธ
                            </button>
                        ` : '<span class="text-gray-500 text-[11px]">เสร็จสิ้น</span>'}
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function updateOrderStatus(orderId, newStatus) {
            const order = orders.find(o => o.id === orderId);
            if (order) {
                order.status = newStatus;
                saveOrdersToStorage(false);
                renderAdminOrders();
                broadcastChange('ORDER_STATUS_CHANGED', { orderId, status: newStatus });
                showToast(`อัปเดตคำสั่งซื้อ #${orderId.slice(-6)} เป็น ${newStatus} แล้ว`, 'info');
            }
        }

        function updatePendingOrdersBadge() {
            const pendingCount = orders.filter(o => o.status === 'Pending').length;
            const badge = document.getElementById('pendingBadge');
            const countDisplay = document.getElementById('orderBadgeCount');
            
            if (pendingCount > 0) {
                badge.innerText = pendingCount;
                badge.classList.remove('hidden');
                countDisplay.innerText = pendingCount;
            } else {
                badge.classList.add('hidden');
                countDisplay.innerText = '0';
            }
        }

        function viewSlipModal(imgSrc) {
            document.getElementById('modalSlipImg').src = imgSrc;
            document.getElementById('slipModal').classList.remove('hidden');
        }

        function closeSlipModal() {
            document.getElementById('slipModal').classList.add('hidden');
        }

        // --- UTILS ---
        function copyToClipboard(text, label) {
            const input = document.createElement('input');
            input.value = text;
            document.body.appendChild(input);
            input.select();
            document.execCommand('copy');
            document.body.removeChild(input);
            showToast(`คัดลอก ${label} (${text}) เรียบร้อยแล้ว`, 'success');
        }

        function showToast(message, type = 'info') {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            
            let bgClass = 'bg-slate-900 border-amber-500/50 text-amber-300';
            let icon = '<i class="fa-solid fa-circle-info text-amber-400"></i>';

            if (type === 'success') {
                bgClass = 'bg-slate-900 border-emerald-500/50 text-emerald-300';
                icon = '<i class="fa-solid fa-circle-check text-emerald-400"></i>';
            } else if (type === 'error') {
                bgClass = 'bg-slate-900 border-red-500/50 text-red-300';
                icon = '<i class="fa-solid fa-triangle-exclamation text-red-400"></i>';
            }

            toast.className = `p-3.5 rounded-xl border shadow-xl ${bgClass} backdrop-blur-md flex items-center gap-3 text-xs pointer-events-auto transform transition-all duration-300 translate-y-2 opacity-0`;
            toast.innerHTML = `${icon} <span class="font-medium flex-1">${message}</span>`;

            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 4000);
        }
